# РАБОТА С СЕРТИФИКАТОМ ПОДПИСИ КОДА
# ═══════════════════════════════════════════════════════════════════════════════
# Все команды выполняются в PowerShell от имени администратора.
# Все файлы сохраняются в: Project\Электронная подпись\


# ───────────────────────────────────────────────────────────────────────────────
# ШАГ 1. СОЗДАНИЕ СЕРТИФИКАТА
# ───────────────────────────────────────────────────────────────────────────────

$cert = New-SelfSignedCertificate `
  -Type CodeSigningCert `
  -Subject "CN=User" `
  -KeyUsage DigitalSignature `
  -FriendlyName "User Code Signing" `
  -CertStoreLocation "Cert:\CurrentUser\My" `
  -HashAlgorithm SHA256 `
  -NotAfter (Get-Date).AddYears(10)

# Параметры которые ты можешь менять:
#
#   -Subject "CN=He11Catnn"
#       Имя сертификата. CN= это обязательный префикс (Common Name).
#       Это то что видно в столбцах "Кому выдан" и "Кем выдан" в certmgr.
#       У самоподписанного сертификата оба столбца всегда одинаковые.
#       Пример: -Subject "CN=МоёПриложение"
#
#   -FriendlyName "He11Catnn Code Signing"
#       Отображаемое имя в certmgr (столбец "Имя").
#       Только косметика, на работу не влияет. Можно написать что угодно.
#       Пример: -FriendlyName "Моя подпись 2025"
#
#   -NotAfter (Get-Date).AddYears(10)
#       Срок действия сертификата. Сейчас стоит 10 лет от даты создания.
#       Можно поставить конкретную дату:
#       -NotAfter "31.12.2040"
#       Или другой срок:
#       -NotAfter (Get-Date).AddYears(5)   # 5 лет
#       -NotAfter (Get-Date).AddMonths(18) # 18 месяцев
#
#   -HashAlgorithm SHA256
#       Алгоритм хэширования. SHA256 — стандарт, не меняй.
#
#   -CertStoreLocation "Cert:\CurrentUser\My"
#       Куда временно помещается сертификат для экспорта. Не меняй.


# ───────────────────────────────────────────────────────────────────────────────
# ШАГ 2. ЭКСПОРТ В .PFX (нужен для подписания файлов)
# ───────────────────────────────────────────────────────────────────────────────

$password = ConvertTo-SecureString -String "<pasword>" -Force -AsPlainText

Export-PfxCertificate `
  -Cert $cert `
  -FilePath "Project\Электронная подпись\User.pfx" `
  -Password $password

# Параметры которые ты можешь менять:
#
#   -String "<pasword>"
#       Пароль которым защищается .pfx файл.
#       Этот же пароль нужно указывать при подписании (параметр /p).
#       Запомни или сохрани — без него .pfx бесполезен.
#
#   -FilePath "..."
#       Путь куда сохранится .pfx файл.
#       Измени имя файла если изменил Subject выше.

Write-Host "Готово! .pfx сохранён: Project\Электронная подпись\User.pfx"
Write-Host "Thumbprint: $($cert.Thumbprint)"

# Thumbprint — уникальный отпечаток сертификата, запиши его.
# Он понадобится если захочешь найти этот сертификат в хранилище позже.


# ───────────────────────────────────────────────────────────────────────────────
# ШАГ 3. ЭКСПОРТ В .CER (нужен для встраивания в установщик)
# ───────────────────────────────────────────────────────────────────────────────

Export-Certificate `
  -Cert $cert `
  -FilePath "Project\Электронная подпись\User.cer"

# .cer — это публичная часть сертификата без пароля.
# Именно он встраивается в установщик Inno Setup и устанавливается
# на компьютер пользователя при инсталляции приложения.
# .pfx содержит приватный ключ и используется только тобой для подписания.


# ───────────────────────────────────────────────────────────────────────────────
# ШАГ 4. РУЧНОЕ ПОДПИСАНИЕ ФАЙЛА (если нужно подписать вручную)
# ───────────────────────────────────────────────────────────────────────────────

& "Windows Kits\10\bin\10.0.26100.0\x64\signtool.exe" sign `
  /f "Project\Электронная подпись\User.pfx" `
  /p <pasword> `
  /fd SHA256 `
  /tr http://timestamp.digicert.com `
  /td SHA256 `
  /v `
  "ПУТЬ_ДО_ФАЙЛА.exe"

# Параметры:
#
#   /f "путь\до\User.pfx"
#       Путь до .pfx файла с приватным ключом.
#
#   /p <pasword>
#       Пароль от .pfx (тот что указывал в шаге 2).
#
#   /fd SHA256
#       Алгоритм хэша для подписи файла. Не меняй.
#
#   /tr http://timestamp.digicert.com
#       Сервер временной метки (timestamp). Он фиксирует время подписания.
#       Это важно: даже когда сертификат истечёт, подпись останется валидной
#       потому что Windows видит что она была сделана пока сертификат был живой.
#       Альтернативные серверы если digicert не отвечает:
#       http://timestamp.sectigo.com
#       http://tsa.starfieldtech.com
#
#   /td SHA256
#       Алгоритм хэша для временной метки. Не меняй.
#
#   /v
#       Verbose — подробный вывод. Убери если хочешь тихий режим.
#
#   "ПУТЬ_ДО_ФАЙЛА.exe"
#       Файл который нужно подписать. Можно указать несколько через пробел.


# ───────────────────────────────────────────────────────────────────────────────
# ШАГ 5. ИЗМЕНИТЬ ОТОБРАЖАЕМОЕ ИМЯ УЖЕ УСТАНОВЛЕННОГО СЕРТИФИКАТА
# ───────────────────────────────────────────────────────────────────────────────

# FriendlyName можно поменять без пересоздания сертификата.
# Запусти в PowerShell от администратора:

$cert = Get-ChildItem Cert:\LocalMachine\Root | Where-Object { $_.Subject -like "*He11Catnn*" }
$cert.FriendlyName = "He11Catnn Code Signing"

# Проверь результат в certmgr.msc — столбец "Имя" обновится.


# ───────────────────────────────────────────────────────────────────────────────
# СПРАВКА: ГДЕ ХРАНЯТСЯ СЕРТИФИКАТЫ
# ───────────────────────────────────────────────────────────────────────────────

# Открыть графический менеджер сертификатов:
#   Win+R → certmgr.msc → Enter
#
# Хранилища которые использует установщик:
#   Доверенные корневые центры сертификации  (LocalMachine\Root)
#       → сюда попадает User.cer при установке
#       → именно здесь ты видел свой сертификат на скриншоте
#
#   Доверенные издатели  (LocalMachine\TrustedPublisher)
#       → убирает предупреждения UAC при запуске подписанных .exe
#
# Проверить через PowerShell:
#   Get-ChildItem Cert:\LocalMachine\Root | Where-Object { $_.Subject -like "*User*" }
#   Get-ChildItem Cert:\LocalMachine\TrustedPublisher | Where-Object { $_.Subject -like "*User*" }
#
# Удалить сертификат из хранилища (если нужно пересоздать):
#   $cert = Get-ChildItem Cert:\LocalMachine\Root | Where-Object { $_.Subject -like "*User*" }
#   Remove-Item $cert.PSPath
